/* ── Starfield ── */
const cvs = document.getElementById('cvs');
const ctx = cvs.getContext('2d');
let W, H, stars = [];
function initStars() {
  W = cvs.width = window.innerWidth;
  H = cvs.height = window.innerHeight;
  stars = Array.from({length:260}, () => ({
    x:Math.random()*W, y:Math.random()*H,
    r:Math.random()*1.1+.15,
    sp:Math.random()*.4+.08,
    ph:Math.random()*Math.PI*2
  }));
}
function drawStars(t) {
  ctx.clearRect(0,0,W,H);
  const g = ctx.createRadialGradient(W*.3,H*.2,0,W*.5,H*.5,Math.max(W,H)*.8);
  g.addColorStop(0,'#1a0a2e'); g.addColorStop(1,'#07050f');
  ctx.fillStyle=g; ctx.fillRect(0,0,W,H);
  stars.forEach(s => {
    const a=.25+.75*Math.abs(Math.sin(t*s.sp*.0008+s.ph));
    ctx.beginPath(); ctx.arc(s.x,s.y,s.r,0,Math.PI*2);
    ctx.fillStyle=`rgba(255,248,200,${a})`; ctx.fill();
  });
  requestAnimationFrame(drawStars);
}
window.addEventListener('resize', initStars);
initStars(); requestAnimationFrame(drawStars);

/* ── Render helpers ── */
function loadingHTML() {
  return `<div class="loading-state">
    <div class="spinner"><span class="spinner-star">✦</span></div>
    <span class="loading-text">Loading content…</span>
  </div>`;

function errorHTML(tabKey) {
  return `<div class="error-state">
    <div class="error-icon">🌙</div>
    <p class="error-msg">Unable to load this content from the server.<br>Please check your connection and try again.</p>
    <button class="retry-btn" onclick="loadTab('${tabKey}', true)">✦ &nbsp;Retry</button>
  </div>`;
}

function renderContent(content) {
  if (!content || !content.trim()) return `<p class="policy-plain" style="text-align:center;opacity:.5">No content available.</p>`;
  const t = content.trim();
  /* If the API returns HTML */
  if (/<[a-z][\s\S]*>/i.test(t)) {
    return `<div class="policy-body">${t}</div>`;
  }
  /* Markdown-ish plain text */
  let html = t
    .replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/^### (.+)$/gm,'<h3>$1</h3>')
    .replace(/^## (.+)$/gm,'<h2>$1</h2>')
    .replace(/^# (.+)$/gm,'<h1>$1</h1>')
    .replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>')
    .replace(/\*(.+?)\*/g,'<em>$1</em>')
    .replace(/^\- (.+)$/gm,'<li>$1</li>')
    .replace(/^(\d+)\. (.+)$/gm,'<li>$2</li>')
    .replace(/(<li>[\s\S]+?<\/li>)(\n(?!<li>)|$)/g,'<ul>$1</ul>')
    .replace(/\n\n+/g,'</p><p>')
    .replace(/\n/g,'<br>');

  const hasFormatting = /^(#{1,3} |\*\*|\- )/m.test(t);
  if (hasFormatting) return `<div class="policy-body"><p>${html}</p></div>`;
  return `<pre class="policy-plain">${t}</pre>`;
}

async function loadTab(tabKey, force = false) {
  const el = document.getElementById('policyContent');

  if (!force && cache[tabKey]) {
    el.innerHTML = renderContent(cache[tabKey]);
    return;
  }

  el.innerHTML = loadingHTML();

  const cfg = TAB_CONFIG[tabKey];
  try {
    const res = await fetch(cfg.url, {
      headers: { 'Accept': 'application/json' }
    });
    const data = await res.json();

    if (data.success && data.content) {
      cache[tabKey] = data.content;
      /* Update date if available */
      if (data.updatedAt || data.updated_at) {
        const d = new Date(data.updatedAt || data.updated_at);
        document.getElementById('metaDate').textContent =
          'Last updated: ' + d.toLocaleDateString('en-IN', {year:'numeric',month:'long',day:'numeric'});
      }
      el.innerHTML = renderContent(data.content);
    } else {
      el.innerHTML = errorHTML(tabKey);
    }
  } catch (err) {
    console.error(`[BK] Failed to load ${tabKey}:`, err);
    el.innerHTML = errorHTML(tabKey);
  }
}

/* ── Switch tab ── */
function switchTab(tabKey) {
  if (activeTab === tabKey) return;
  activeTab = tabKey;

  /* Update tab buttons */
  document.querySelectorAll('.tab').forEach(btn => {
    btn.classList.toggle('active', btn.dataset.tab === tabKey);
  });

  /* Update hero & meta */
  const cfg = TAB_CONFIG[tabKey];
  document.getElementById('heroTitle').innerHTML     = cfg.title;
  document.getElementById('heroSub').textContent     = cfg.titleBn;
  document.getElementById('heroEyebrow').textContent = cfg.eyebrow;
  document.getElementById('metaTag').innerHTML       = cfg.tag;
  document.getElementById('breadcrumbCurrent').innerHTML = cfg.breadcrumb;
  document.getElementById('metaDate').textContent    = 'Bhavishya Katha · 2026';

  /* Update page title */
  document.title = `${cfg.title} | Bhavishya Katha`;

  /* Load content */
  loadTab(tabKey);

  /* Scroll to tabs */
  document.querySelector('.tabs-wrap').scrollIntoView({ behavior: 'smooth', block: 'start' });
}

/* ── Init: load first tab ── */
switchTab('user-terms');
