/* Starfield */
const cvs = document.getElementById('cvs');
const ctx = cvs.getContext('2d');
let W, H, stars = [];

function init() {
  W = cvs.width  = window.innerWidth;
  H = cvs.height = window.innerHeight;
  stars = Array.from({length: 260}, () => ({
    x: Math.random() * W, y: Math.random() * H,
    r: Math.random() * 1.1 + 0.15,
    sp: Math.random() * 0.4 + 0.08,
    ph: Math.random() * Math.PI * 2,
  }));
}

function draw(t) {
  ctx.clearRect(0, 0, W, H);
  const g = ctx.createRadialGradient(W*.3,H*.2,0,W*.5,H*.5,Math.max(W,H)*.8);
  g.addColorStop(0,'#1a0a2e'); g.addColorStop(1,'#07050f');
  ctx.fillStyle = g; ctx.fillRect(0,0,W,H);
  stars.forEach(s => {
    const a = .25 + .75 * Math.abs(Math.sin(t * s.sp * .0008 + s.ph));
    ctx.beginPath(); ctx.arc(s.x, s.y, s.r, 0, Math.PI*2);
    ctx.fillStyle = `rgba(255,248,200,${a})`; ctx.fill();
  });
  requestAnimationFrame(draw);
}

window.addEventListener('resize', init);
init(); requestAnimationFrame(draw);

/* Hamburger */
const tog = document.getElementById('navToggle');
const nl  = document.getElementById('navLinks');
tog.addEventListener('click', () => {
  tog.classList.toggle('open');
  nl.classList.toggle('open');
  document.querySelector('.float-app').style.display = nl.classList.contains('open') ? 'none' : 'flex';
});
nl.querySelectorAll('a').forEach(a => a.addEventListener('click', () => {
  tog.classList.remove('open'); nl.classList.remove('open');
  document.querySelector('.float-app').style.display = 'flex';
}));

/* Scroll reveal */
const io = new IntersectionObserver(es => {
  es.forEach(e => { if(e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target); } });
}, { threshold:.1 });
document.querySelectorAll('.reveal').forEach(el => io.observe(el));
